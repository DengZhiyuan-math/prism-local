\begin{definition}\label{def:convex}
  A function $f\colon \R^n \to \R$ is \emph{convex} if
  \begin{equation}\label{eq:convex}
    f(t x + (1-t) y) \le t f(x) + (1-t) f(y)
  \end{equation}
  for all $x, y \in \R^n$ and $t \in [0,1]$.
\end{definition}

\begin{lemma}\label{lem:norm-convex}
  Every norm $\norm{\cdot}$ on $\R^n$ is convex in the sense of \cref{def:convex}.
\end{lemma}

\begin{proof}
  By the triangle inequality and homogeneity,
  $\norm{t x + (1-t) y} \le t\norm{x} + (1-t)\norm{y}$, which is \eqref{eq:convex}.
\end{proof}
